reformation.2 = {
	type = country_event
	category = situation_event

	title = reformation.2.title
	desc = reformation.2.desc
	outcome = neutral


	option = {
		name = reformation.2.a
		add_country_modifier = {
			modifier = bible_translations_banned
			mode = add_and_extend
			years = 25
		}
		
		ai_will_select  = {
			if = {
				limit = {
					OR = {
						AND = {
							capital.region = region:italy_region
							culture = { has_culture_group = culture_group:italian_group }
							OR = {
								location:rome = {
									owner ?= c:PAP
								}
								c:PAP = {
									num_locations_owned_or_owned_by_subjects_or_below > 4
								}
							}
						}
						AND = {
							capital.region = region:iberia_region
							culture = { has_culture_group = culture_group:iberian_group }
						}
						capital.region = region:ireland_region
						culture = culture:danube_bavarian
					}
				}
				value = 50000
			}
			if = {
				limit = {
					government_type = government_type:theocracy
				}
				value = 40000
			}
		}
	}

	option = {
		name = reformation.2.b

		add_religious_influence = religious_influence_weak_penalty

		# Bibles in the vernacular seed Lutheran converts directly
		ordered_owned_location = {
			order_by = population
			limit = { dominant_religion = religion:catholic }
			max = {
				value = 1
				add = {
					value = root.num_provinces
					multiply = 0.25
				}
			}
			check_range_bounds = no
			every_pop = {
				limit = { religion = religion:catholic }
				split_pop = {
					fraction = 0.05
					religion = religion:lutheran
				}
			}
		}
		
		ai_will_select  = {
			if = {
				limit = {
					OR = {
						culture = { has_culture_group = culture_group:french_group }
						culture = { has_culture_group = culture_group:netherlandish_group }
						has_or_had_tag = FRA
						has_or_had_tag = HUN
					}
				}
				value = -500000
			}
		}
	}
	
	option = {
		name = reformation.2.c
		
		trigger = {
			has_global_variable = icd_calvin_here
		}

		add_religious_influence = religious_influence_weak_penalty

		# Bibles in the vernacular seed calvinists converts directly
		ordered_owned_location = {
			order_by = population
			limit = { dominant_religion = religion:catholic }
			max = {
				value = 1
				add = {
					value = root.num_provinces
					multiply = 0.25
				}
			}
			check_range_bounds = no
			every_pop = {
				limit = { religion = religion:catholic }
				split_pop = {
					fraction = 0.05
					religion = religion:calvinist
				}
			}
		}
		
		ai_will_select  = {
			if = {
				limit = {
					OR = {
						culture.dialect = dialect:low_german_dialect						  
						culture = culture:saxon
					}
				}
				value = -5000
			}
			if = {
				limit = {
					NOT = {
						capital.region = region:ireland_region
					}
					OR = {
						culture = { has_culture_group = culture_group:netherlandish_group }
						culture = { has_culture_group = culture_group:british_group }
						culture = { has_culture_group = culture_group:carpathian_group }
						culture = culture:greater_polish
						has_or_had_tag = SCO
					}
				}
				value = 50000
			}
		}
	}
}


reformation.7 = { #John Calvin
	type = country_event
	category = situation_event
	title = reformation.7.title
	desc = reformation.7.desc
	outcome = neutral

	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }

		create_character = {
			first_name = name_john
			last_name = calvin
			culture = culture:picard
			birth_location = location:noyon
			religion = religion:calvinist
			script = john_calvin
			birth_date = 1509.7.10
			estate = estate_type:clergy_estate
			adm = { 40 60 }
			dip = { 40 60 }
			mil = { 80 100 }
			save_scope_as = target_character
		}
		add_to_reformer_list = yes
		situation:reformation = {
			var:lutheran_character = {
				save_scope_as = target_luther
			}
		}
		location:basel = {
			save_scope_as = target_location
		}
	}

	option = {
		name = reformation.7.a

		set_global_variable = icd_calvin_here

		# Spawn the Calvinist movement at Basel with Calvin's initial converts as seed supporters
		
		if = {
			limit = {
				religion:calvinist = {
					is_religion_enabled = no
				}
			}
			religion:calvinist = { enable_religion = yes }
		}

		# Calvin becomes the first spreader of the Calvinist movement

		scope:target_location.owner = {
			trigger_event_silently = {
				id = calvinist.2
				months = { 1 3 }
			}
		}
		religion:calvinist = {
			ordered_movement_in_religion = {
				max = 1
				order_by = { value = 1 }
				add_spreader = {
					character = scope:target_character
					location = scope:target_location
				}
			}
		}
		location:paris.owner = {
			trigger_event_silently = {
				id = calvinist.1
				months = { 1 3 }
			}
		}
		set_variable = received_reformer
	}
}

	}

		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }

		create_character = {
			first_name = name_ulrick # alternative Huldrych
			last_name = Zwingli
			religion = religion:catholic
			adm = 78
			dip = 82
			mil = { 80 100 }
			birth_date = 1484.1.1
			birth_location = location:werdenberg
			culture = culture:high_alemannic
			estate = estate_type:peasants_estate
			save_scope_as = target_character
			script = swi_huldrych_zwingli
		}
		add_to_reformer_list = yes
		situation:reformation = {
			var:lutheran_character = {
				save_scope_as = target_luther
			}
		}
		location:bern = {
			save_scope_as = target_location
		}
	}

	option = {
		name = reformation.3.a
		
		religion:calvinist = {
			enable_religion = yes
		}
		
		scope:target_location = {
			spawn_movement = {
				movement_definition = movement_definition:calvinism_movement
				supporters = {
					value = population
					multiply = 0.15
				}
			}
		}

		religion:calvinist = {
			ordered_movement_in_religion = {
				max = 1
				order_by = { value = 1 }
				add_spreader = {
					character = scope:target_character
					location = scope:target_location
				}
			}
		}

		if = {
			limit = {
				is_swiss_tag = no
			}
			scope:target_character = {
				change_character_religion = religion:calvinist
			}
			province_definition:waldstatte_province = {
				every_location_in_province_definition = {
					every_pop = {
						limit = {
							religion = religion:catholic
							culture = culture:high_alemannic
						}
						split_pop = {
							fraction = 0.1
							religion = religion:calvinist
						}
					}
				}
			}
			province_definition:thurgau_province = {
				every_location_in_province_definition = {
					every_pop = {
						limit = {
							religion = religion:catholic
							culture = culture:high_alemannic
						}
						split_pop = {
							fraction = 0.1
							religion = religion:calvinist
						}
					}
				}
			}
		}
		set_variable = received_reformer
	}
}
