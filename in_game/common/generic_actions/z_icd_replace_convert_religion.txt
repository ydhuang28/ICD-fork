REPLACE:convert_religion = {
	type = owncountry

	sound = UI_action_religion_generic	
	
	ai_tick = monthly
	ai_tick_frequency = 12

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = religiousdoctrines

	
	ai_prerequisite = {
		trigger_if = {
			limit = {
				situation:hussite_wars = {
					situation_is_active = yes
				}
				religion = religion:catholic
			}
			religion_percentage_in_country = {
				religion = religion:hussite
				value > 0.2
			}
		}
		trigger_else_if = {
			limit = {
				situation:reformation = {
					situation_is_active = yes
				}
				situation:council_of_trent = {
					situation_has_ended = no
				}
				religion = religion:catholic
				OR = {
					AND = {
						is_member_of_international_organization = international_organization:hre
						has_presence_in = region:north_german_region
					}
					has_presence_in = region:scandinavian_region
				}
			}
			OR = {
				religion_percentage_in_country = {
					religion = religion:lutheran
					value > 0.2
				}
				religion_percentage_in_country = {
					religion = religion:calvinist
					value > 0.2
				}
				dominant_religion ?= {
					NOT = { this = root.religion }
				}
			}
		}
		trigger_else_if = {
			limit = {
				situation:reformation = {
					situation_is_active = yes
				}
				current_year < 1600
				OR = {
					has_or_had_tag = SCO
					has_or_had_tag = WUR
					AND = {
						is_member_of_international_organization = international_organization:hre
						religion = religion:lutheran
					}
					AND = {
						OR = {
							culture = { has_culture_group = culture_group:lithuanian_group }
							has_or_had_tag = LIT
							has_or_had_tag = POL
						}
						religion = religion:catholic
						current_year > 1560
						OR = {
							AND = {
								country_exists = c:PRU
								c:PRU = {
									religion = religion:catholic
								}
							}
							AND = {
								NOT = {
									country_exists = c:PRU
								}
								c:TEU = {
									religion = religion:catholic
								}
							}
						}
					}
					government_type ?= government_type:republic
				}
			}
			OR = {
				religion_percentage_in_country = {
					religion = religion:lutheran
					value > 0.2
				}
				religion_percentage_in_country = {
					religion = religion:calvinist
					value > 0.2
				}
			}
		}
		trigger_else_if = {
			limit = {
				tag = BOH
				religion != religion:hussite
				OR = {
					situation:hussite_wars = {
						situation_is_active = no
						situation_has_ended = no
					}
					situation:hussite_wars = {
						situation_is_active = yes
					}
				}
			}
			religion_percentage_in_country = {
				religion = religion:hussite
				value > 0.2
			}
		}
		trigger_else = {
			dominant_religion ?= {
				NOT = { this = root.religion }
			}
		}
	}

	potential = {
	}

	allow = {
		trigger_if = {
			limit = {
				exists = scope:actor
			}
			scope:actor = {
				modifier:blocked_from_conversion = no
				is_colonial_subject = no
				custom_tooltip = {
					text = theocracy_subject_cannot_convert_tt
					NOT = {
						AND = {
							government_type = government_type:theocracy
							is_subject = yes
							overlord ?= { government_type = government_type:theocracy }
						}
					}
				}
			}
		}
	}

	price = price:convert_religion

	cooldown = {
		type = convert_religion
		years = 25
	}

	select_trigger = {
		looking_for_a = religion
		target_flag = target
		name = "convert_religion_select_religion"
		none_available_msg_key = "no_convert_religion_tt"
		column = {
			data = name
		}
		column = {
			data = religion_percentage_in_country
		}
		visible = {
			trigger_if = {
				limit = {
					has_game_rule = dynamic_religion
				}
				not = { religious_head ?= scope:actor }
				is_religion_enabled = yes
				not = {
					this = scope:actor.religion
				}
			}
			trigger_else_if = {
				limit = {
					has_game_rule = religion_group_only
				}
				group = scope:actor.religion.group
			}
			trigger_else_if = {
				limit = {
					has_game_rule = religion_present_in_country
				}
				scope:actor = {
					religion_population_in_country = {
						religion = root
						value > 0
					}
				}
			}
			trigger_else = {
				always = yes
			}
			trigger_if = {
				limit = { scope:actor = { is_emperor = yes } }
				group = scope:actor.religion.group
			}
		}
		enabled = {
			custom_tooltip = {
				text = convert_religion_tt
				trigger_if = {
					limit = { exists = root }
					not = { scope:actor.religion = root }
					OR = {
						NOT = { has_game_rule = dynamic_religion }
						scope:actor = {
							religion_population_in_country = {
								religion = root
								value > 0
							}
							or = {
								religion.group = root.group
								religion_percentage_in_country = {
									religion = root
									value > 0.4
								}
							}
						}
					}
				}
				trigger_else = {
					hidden_trigger = { always = yes }
				}
			}
		}
	}

	effect = {
		scope:actor = {
			every_related_country = {
				type = alliance
				limit = {
					this != scope:actor
					religion = scope:actor.religion
				}
				remove_relation = {
					type = relation_type:alliance
					first = this
					second = scope:actor
				}
			}
			if = {
				limit = {
					exists = scope:target
				}
				change_religion = scope:target
				change_religion_for_ruler_and_family = { country = scope:actor religion = scope:target }
			}
			else = {
				custom_tooltip = convert_religion_effect_tt
			}
			
			ruler ?= { remove_character_modifier = enacted_placitum_regium }
		}
	}

	ai_will_do = {

		if = {	#BOH pre-target: general will-to-convert bonus
			limit = {
				scope:actor = c:BOH
				OR = {
					scope:actor = {
						religion != religion:hussite
						religion_percentage_in_country = {
							religion = religion:hussite
							value > 0.2
						}
					}
					situation:hussite_wars = {
						situation_is_active = no
						situation_has_ended = no
					}
				}
			}
			add = {
				value = 250
				desc = "DIPLOREASON_HISTORICAL_DECISION"
			}
		}
		if = {	#BOH per-target: steer specifically towards hussite
			limit = {
				scope:actor = c:BOH
				exists = scope:target
				scope:target = religion:hussite
				scope:actor = {
					religion != religion:hussite
					religion_percentage_in_country = {
						religion = religion:hussite
						value > 0.2
					}
				}
			}
			add = {
				value = 500
				desc = "DIPLOREASON_HISTORICAL_DECISION"
			}
		}
		if = {
			limit = {
				scope:actor = {
					has_or_had_tag = TUR
				}
				OR = {
					has_game_rule = historical_ai_choices
					scope:actor = {
						culture = culture:turkish_culture
						NOT = {
							any_owned_location = {
								any_pop = {
									culture = culture:turkish_culture
									religion.group != religion_group:muslim
									percent = 0.15
								}
							}
						}
					}
				}
			}
			subtract = {
				value = 10000
				desc = "DIPLOREASON_HISTORICAL_DECISION"
			}
		}

		if = {
			limit = {
				has_game_rule = only_within_religion_group
				scope:actor.religion.group != scope:target.group
				trigger_if = {
					limit = { scope:actor = { tag = LIT } }	#Lithuania's conversion to Christianity should be still happening
					scope:target.group = religion_group:christian
				}
			}
			subtract = {
				value = 10000
				desc = "ANTAGONISM_OTHERRELIGION_GROUP"
			}
		}

		if = {
			limit = { scope:actor.dominant_religion = scope:target }
			add = {
				desc = "[dominant_religion|e]"
				value = 250
				multiply = "scope:actor.religion_percentage_in_country(scope:target)"
			}
		}
		if = {
			limit = {
				#check non-protestant religions and see if our ruler is not of that religion 
				#take Catholic Lithuania with Orthodox majority as example
				scope:target = { ai_wants_convert = no }
				scope:actor.dominant_religion = scope:target	
				scope:actor.ruler_or_regent ?= { religion != scope:target }
				scope:actor != c:BOH
			}
			subtract = {
				desc = "DIPLOREASON_RULER_RELIGION"
				value = 1000
			}
		}
		if = {
			limit = {
				scope:actor ?= {
					religion = religion:catholic
					OR = {
						AND = {
							is_member_of_international_organization = international_organization:hre
							has_presence_in = region:north_german_region
						}
						has_presence_in = region:scandinavian_region
					}
				}
			}
			add = {
				desc = "DIPLOREASON_TARGET_RELIGION_PERCENTAGE"
				value = 800
				multiply = "scope:actor.religion_percentage_in_country(scope:target)"
			}
		}
		if = {
			limit = {
				scope:actor = { is_subject = yes }
				scope:target != scope:actor.overlord.religion
			}
			subtract = {
				desc = "DIPLOREASON_OVERLORD_RELIGION"
				value = 1000
			}
		}
		if = { ##swiss that denied reformed will refuse###
			limit = {
				scope:actor = { 
					has_variable = icd_swi_refused_calvinism 
					religion = religion:catholic
				}
			}
			subtract = {
				value = 1000
			}
		}
		if = { ##italians under papal influence will refuse###
			limit = {
				c:PAP = {
					owns = location:rome
				}
				scope:actor ?= {
					capital.region = region:italy_region
					culture = { has_culture_group = culture_group:italian_group }
					religion = religion:catholic
				}
			}
			subtract = {
				value = 10000
			}
		}
		if = { ##bishoprics will refuse###
			limit = {
				scope:actor ?= {
					government_type = government_type:theocracy
					OR = {
						num_locations_owned_or_owned_by_subjects_or_below > 2
						has_or_had_tag = ITEUT
					}
					religion = religion:catholic
					OR = {
						"religion_percentage_in_country(scope:actor.religion)" >= 0.1
						has_or_had_tag = ITEUT
					}
				}
			}
			subtract = {
				value = 4000000
			}
		}
		if = { ##england will refuse unless ruler is of other religion###
			limit = {
				exists = scope:target
				scope:actor = {
					OR = {
						has_or_had_tag = ENG
						has_or_had_tag = GBR
					}
					religion = religion:anglican
					ruler ?= {
						religion = religion:anglican
					}
				}
				scope:target = religion:catholic
			}
			subtract = {
				value = 2000
			}
		}
		if = { ###turkics will not convert to tengri usually###
			limit = {
				exists = scope:target
				scope:actor ?= {
					OR = {
						has_or_had_tag = GLH
						culture = { has_culture_group = culture_group:turkic_group }
						culture = { has_culture_group = culture_group:mongolian_group }
					}
					OR = {
						religion.group = religion_group:muslim
						religion.group = religion_group:buddhist
						religion.group = religion_group:christian
					}
					OR = {
						government_type = government_type:steppe_horde
						government_type = government_type:tribe
					}
					"religion_percentage_in_country(scope:actor.religion)" > 0.1
				}
				scope:target = religion:tengri
			}
			subtract = {
				value = 2000
			}
		}
		if = { ##some countries will want calvinism### 
			limit = {
				exists = scope:target
				scope:actor ?= {
					OR = {
						AND = {
							has_or_had_tag = SCO
							religion = religion:catholic
						}
						AND = {
							culture = { has_culture_group = culture_group:netherlandish_group }
							OR = {
								religion = religion:lutheran
								religion = religion:catholic
							}
						}
						AND = {
							is_member_of_international_organization = international_organization:hre
							religion = religion:lutheran	
						}
					}
					"religion_percentage_in_country(religion:calvinist)" >= 0.2
				}
				scope:target = religion:calvinist
			}
			add = {
				value = 900
			}
		}	
		if = { ##hre emperor will usually refuse### 
			limit = {
				scope:actor ?= {
					religion = religion:catholic
					OR = {
						is_leader_of_international_organization = international_organization:hre
						any_union_partner = {
							is_leader_of_international_organization = international_organization:hre
							religion = religion:catholic
						}
					}
					"religion_percentage_in_country(religion:catholic)" >= 0.05
				}
			}
			subtract = {
				value = 2000
			}
		}
		if = { ##emperor of china### 
			limit = {
				scope:actor ?= {
					religion = religion:sanjiao
					is_leader_of_international_organization = international_organization:middle_kingdom
				}
			}
			subtract = {
				value = 400000
			}
		}
		if = { ##bavarians will avoid a little### 
			limit = {
				scope:actor ?= {
					culture = culture:danube_bavarian
					religion = religion:catholic
				}
				OR = {
					scope:target = religion:calvinist
					scope:target = religion:lutheran
				}
			}
			subtract = {
				value = 120
			}
		}
		if = { ##stop pedalling either way### 
			limit = {
				exists = scope:target
				current_year > 1565
				scope:actor ?= {
					NOT = { culture.dialect = dialect:low_german_dialect }
					OR = {
						religion = religion:catholic
						religion = religion:lutheran
						religion = religion:calvinist
					}
					is_member_of_international_organization = international_organization:hre
				}
				OR = {
					scope:target = religion:catholic
					scope:target = religion:lutheran
					scope:target = religion:calvinist
				}
			}
			subtract = {
				value = 5000000
			}
		}
		if = { ##lithuanians interested if teutons refuse### 
			limit = {
				exists = scope:target
				scope:actor ?= {
					OR = {
						culture = { has_culture_group = culture_group:lithuanian_group }
						has_or_had_tag = LIT
						has_or_had_tag = POL
					}
					religion = religion:catholic
				}
				current_year > 1560
				OR = {
					AND = {
						country_exists = c:PRU
						c:PRU = {
							religion = religion:catholic
						}
					}
					AND = {
						NOT = {
							country_exists = c:PRU
						}
						c:TEU = {
							religion = religion:catholic
						}
					}
				}
				OR = {
					AND = {
						scope:target = religion:calvinist
						scope:actor = {
							religion_percentage_in_country = {
								religion = religion:calvinist
								value > 0.2
							}
						}
					}
					AND = {
						scope:target = religion:lutheran
						scope:actor = {
							religion_percentage_in_country = {
								religion = religion:lutheran
								value > 0.2
							}
						}
					}
				}
			}
			add = {
				value = 500000
			}
		}
		if = { ##muslim conquerors shouldnt always adopt hindu###
			limit = {
				scope:actor = {  
					religion.group = religion_group:muslim
				}
				scope:target = religion:hindu
			}
			subtract = {
				value = 500000
			}
		}
	}
}
