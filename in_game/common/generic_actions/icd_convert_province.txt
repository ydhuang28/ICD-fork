icd_convert_province = {	
	type = religious

	sound = UI_action_religion_generic
	
	potential = {
		scope:actor = {
			num_locations > 0
			NOT = {
				has_policy = icd_sulh_i_kul
				has_policy = icd_ethnic_religion
				has_policy = icd_custodians
				has_policy = icd_secular_state
				has_policy = icd_yassa_tolerance
				has_policy = icd_dual_faiths
			}
			hidden_trigger = {
				OR = {
					icd_is_pagan = no
					is_ai = no
				}
			}
		}
		NOT = { has_game_rule = icd_religious_conversion_vanilla }
		
	}
	
	allow = {
		scope:actor = {
			religious_unity < 1.0
			NOT = {
				AND = {
					modifier:global_nobles_conversion_blocked = yes
					modifier:global_burghers_conversion_blocked = yes
					modifier:global_clergy_conversion_blocked = yes
					modifier:global_peasants_conversion_blocked = yes
				}
			}
			custom_tooltip = {
				text = icd_convert_province_actions_tt
				icd_num_provinces_for_conversion >= 1
			}
		}
	}
	
	ai_tick = monthly
	ai_tick_frequency = 6
		
	maximum_targets_in_one_tick = 15

	select_trigger = {
		looking_for_a = religion
		source = actor
		target_flag = recipient
		name = "choose_religion"
		column = {
			data = name
		}
	}

	select_trigger = {
		looking_for_a = religion
		target_flag = target_religion
		source_ai_override = actor
		source_flags_ai_override = include_any_present
		name = "promote_religion_select_religion"
		none_available_msg_key = "convert_religion_no_religions"
		column = {
			data = name
		}
		column = {
			data = religion_percentage_in_country
		}
		default_sort = religion_percentage_in_country
		visible = {
			OR = {
				AND = {
					scope:actor = { is_ai = no }
					group = scope:actor.religion.group
					is_religion_enabled = yes
					OR = {
						scope:actor.religion = root
						scope:actor = {
							religion_percentage_in_country = {
								religion = root
								value >= 0.01
							}
						}
					}
				}
				AND = {
					scope:actor = { is_ai = yes }
					scope:actor.religion = root
				}
			}
		}
	}
	
	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_province
		name = "promote_religion_select_province"
		none_available_msg_key = "promote_religion_no_provinces"
		column = {
			data = name
		}
		column = {
			data = province_religion_percentage
		}
		column = {
			data = population
		}
		default_sort = province_religion_percentage
		visible = {
			NOT = {
				has_province_modifier = icd_promote_religion_province_modifier
			}
		}
		show_why_not_enabled = yes
		enabled = {
			any_location_in_province = {
				OR = {
					NOT = { dominant_religion = owner.religion }
					local_religious_unity < 1.0
				}
			}
			any_location_in_province = {
				any_pop = {
					custom_tooltip = {
						text = icd_convert_province_pop_tt
						religion != scope:target_religion
						religion != scope:actor.religion
						pop_type != pop_type:slaves
						is_protected_pop = no
						owner ?= scope:actor
					}
				}
				NOT = {
					owner ?= {
						has_estate_privilege = estate_privilege:icd_druze_emirs
					}
					OR = {
						province_definition = province_definition:lebanon_province
						province_definition = province_definition:hims_province
						province_definition = province_definition:hawran_province
						province_definition = province_definition:shaam_province
						province_definition = province_definition:latakia_province
					}
				}
				NOT = {
					has_location_modifier = icd_religious_zeal
				}
			}
			NOT = {
				has_province_modifier = icd_promote_religion_province_modifier
			}
		}
		selected = {
			has_province_modifier = icd_promote_religion_province_modifier
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = "province.religion_percentage(scope:target_religion)"
			}
		}
	}
	
	show_message = no
	
	effect = {
		if = {
			limit = {
				exists = scope:target_province
				scope:actor = { is_ai = yes }
			}
		scope:target_province = {
				set_variable = {
					name = promoting_religion
					value = scope:target_religion
					months = 48
				}
				add_province_modifier = {
					modifier = icd_promote_religion_province_modifier
					mode = add_and_extend
					months = 48
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:target_province
				scope:actor = { is_ai = no }
			}
			scope:target_province = {
				set_variable = {
					name = promoting_religion
					value = scope:target_religion
				}
				add_province_modifier = {
					modifier = icd_promote_religion_province_modifier
					mode = add_and_extend
				}
			}
		}
	}
	
	ai_will_do = {
		value = 0
		
		add = {
			desc = "Wrong religion"
			if = {
				limit = {
					exists = scope:target_religion
					scope:target_religion != scope:actor.religion				
				}
				value = -1000
			}
		}
		
		if = {
			limit = {
				scope:actor = {
					OR = {
						has_or_had_tag = POR
						has_or_had_tag = CAS
						has_or_had_tag = SPA
						has_or_had_tag = GRA
						has_or_had_tag = ARA
						has_or_had_tag = FRA
						has_or_had_tag = HAB
						culture = { has_culture_group = culture_group:russian_group }
						culture = culture:danube_bavarian
						has_or_had_tag = TIM
					}
				}
			}
			add = {
				value = 50
			}
		}

		if = {
			limit = {
				scope:actor = {
					has_policy = self_governing_institutions
				}
			}
			add = {
				value = -10
			}
		}
	
		if = {
			limit = {
				scope:actor = {
					is_colonial_subject = yes
				}
			}
			add = {
				value = 150
			}
		}
		
		if = {
			limit = {
				scope:actor = {
					ruler ?= {
						has_trait = tolerant
					}
				}
				OR = {
					scope:actor = { NOT = { can_see_situation = situation:reformation } }
					situation:reformation = {
						situation_is_active = no
					}
					NOT = {
						scope:actor = {
							religion = religion:catholic
							religion = religion:lutheran
							religion = religion:calvinist
							religion = religion:anabaptist
							religion = religion:unitarian
						}
					}
				}
			}
			add = {
				value = -200
			}
		}
		if = {
			limit = {
				scope:actor = {
					ruler ?= {
						has_trait = zealot
					}
				}
			}
			add = {
				value = 15
			}
		}
		if = {
			limit = {
				scope:actor = {
					religious_unity < 0.5
				}
			}
			add = {
				value = 10
			}
		}
		if = {
			limit = {
				scope:actor = {
					societal_value:spiritualist_vs_humanist < -50
				}
			}
			add = {
				value = 10
			}
		}
		if = {
			limit = {
				scope:actor = {
					OR = {
						religion.group = religion_group:christian
						religion.group = religion_group:muslim
					}
					any_pop = {
						religion != scope:actor.religion
						is_protected_pop = no
					}
				}
			}
			add = {
				value = 10
			}
		}
		if = {
			limit = {
				scope:actor = {
					can_see_situation = situation:reformation
					situation:reformation = {
						situation_is_active = yes
					}
				}
			}
			add = {
				value = 320
			}
		}
		if = {
			limit = {
				scope:actor = {
					has_reform = government_reform:partitio_reform
				}
			}
			add = {
				value = -70
			}
		}
		if = {
			limit = {
				scope:actor = {
					any_owned_location = {
						any_pop = {
							icd_is_pagan = yes
							pop_type != pop_type:slaves
							is_protected_pop = no
							owner ?= scope:actor
						}
					}
				}
			}
			add = {
				value = 20
			}
		}
		if = {
			limit = {
				scope:actor = {
					government_type = government_type:republic
				}
			}
			add = {
				value = -10
			}
		}
		if = {
			limit = {
				scope:actor = {
					government_type = government_type:theocracy
				}
			}
			add = {
				value = 10
			}
		}
	
		if = {
			limit = { exists = scope:target_province }
			
			add = {
				value = -150
			}
			
			add = {
				value = 10
				multiply = scope:target_province.province_average_control
			}
			subtract = {	
				value = 10
				multiply = "scope:target_province.religion_percentage(scope:target_religion)"
			}
			if = {
				limit = {
					scope:target_province = {
						is_overseas_for_owner = no
					}
				}
				add = {	
					value = 5
				}
			}
			
			if = {
				limit = {
					scope:target_province = {
						province_average_control < 0.1
					}
				}
				subtract = {	
					value = 500
				}
			}
			
			if = {
				limit = {
					scope:actor = {
						can_see_situation = situation:reformation
						situation:reformation = {
							situation_is_active = yes
						}
					}
					scope:target_province = {
						OR = {
							dominant_religion = religion:orthodox
							dominant_religion = religion:sunni
							dominant_religion = religion:shia
						}
					}
				}
				subtract = {	
					value = 1
				}
			}
			
			if = { ## ONLY pagans and abrahamic###
				limit = {
					scope:actor = {
						OR = {
							religion.group = religion_group:muslim
							religion.group = religion_group:christian
							religion.group = religion_group:zoroastrian_group
						}
					}
					scope:target_province = {
						any_location_in_province = {
							owner ?= scope:actor
							any_pop = {
								pop_type != pop_type:slaves
								is_protected_pop = no
								icd_is_pagan = yes
								owner ?= scope:actor
							}
							NOT = {
								any_pop = {
									icd_is_pagan = no
									religion != scope:target_religion
								}
							}
						}
					}
				}
				add = {
					value = 2
				}
			}
			
			if = { ## enemy religion###
				limit = {
					scope:target_province = {
						any_location_in_province = {
							owner ?= scope:actor
							any_pop = {
								pop_type != pop_type:slaves
								is_protected_pop = no
								owner ?= scope:actor
								religion = { reverse_religious_view = { target = scope:actor.religion value = enemy } }
								religion != scope:actor.religion
							}
							"religion_percentage(scope:actor.religion)" < 0.95
						}
					}
				}
				add = {
					value = 250
				}
			}
			
			if = { ## negative religion###
				limit = {
					scope:target_province = {
						any_location_in_province = {
							owner ?= scope:actor
							any_pop = {
								pop_type != pop_type:slaves
								owner ?= scope:actor
								is_protected_pop = no
								religion = { reverse_religious_view = { target = scope:actor.religion value = negative } }
								religion != scope:actor.religion
							}
							"religion_percentage(scope:actor.religion)" < 0.95
						}
					}
				}
				add = {
					value = 200
				}
			}
			if = { ## neutral religion - christian and muslims only###
				limit = {
					scope:actor ?= {
						OR = {
							religion.group = religion_group:christian
							religion.group = religion_group:muslim
						}
					}
					scope:target_province = {
						any_location_in_province = {
							owner ?= scope:actor
							any_pop = {
								pop_type != pop_type:slaves
								is_protected_pop = no
								owner ?= scope:actor
								religion != religion:yazidism
								religion = { reverse_religious_view = { target = scope:actor.religion  value = neutral } }
								religion != scope:actor.religion
							}
							"religion_percentage(scope:actor.religion)" < 0.95
						}
					}
				}
				add = {
					value = 200
				}
			}
			
			
			if = {
				limit = {
					scope:target_province = {
						any_location_in_province = {
							has_location_modifier = icd_religious_zeal
							percent = 0.8
						}
					}
				}
				add = {
					value = -2000
				}
			}
		}
	}
}
