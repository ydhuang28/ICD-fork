introduce_cultural_settlements = {
	ability = adm
	
	allow_multiple = yes
	
	potential = {
		num_locations > 0 
	}
	
	allow = {
		cultural_unity < 1.0
	}
	
	select_trigger = {
		looking_for_a = culture
		target_flag = target
		#for the AI we have an optimisation to only look through religions that are present. Human can look at all...but will have some disabled
		source_ai_override = actor
		source_flags_ai_override = include_any_present
		name = "promote_culture_select_culture"
		none_available_msg_key = "promote_culture_no_culture"
		column = {
			data = name
		}
		column = {
			data = culture_percentage_in_country
		}
		default_sort = culture_percentage_in_country
		visible = {
			OR = {
				AND = { # for humans
					scope:actor = {
						is_ai = no
					}
					is_primary_or_accepted_in = scope:actor
				}

				AND = { # for AI
					scope:actor = {
						is_ai = yes
					}
					is_primary_in = scope:actor 
				}
			}
			scope:actor = {
				any_owned_location = {
					dominant_culture = root
				}
			}
		}
	}
	
	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_origin
		name = "relocate_people_select_origin_province"
		none_available_msg_key = "relocate_people_no_provinces"
		column = {
			data = name
		}
		column = {
			data = population
		}
		column = {
			data = province_culture_percentage
		}
		default_sort = population
		visible = {
			culture_population = {
				culture = scope:target
				value > 0.500
			}
			# Several copies can run, each from its own origin to its own destination.
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = introduce_cultural_settlements interaction_target = target_origin } }
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = relocate_people interaction_target = target_origin } }
		}
		map_color = define:NMapColors|MAP_COLOR_HIGH
	}
	
	select_trigger = {
		looking_for_a = pop_type
		source = actor
		target_flag = target_pop
		name = "relocate_people_select_pop"
		none_available_msg_key = "relocate_people_no_pops"
		column = {
			data = name
		}
		default_sort = population
		visible = {
			OR = {
				AND = {
					root = pop_type:peasants
					scope:target_origin = {
						any_location_in_province = {
							any_pop = {
								pop_type = pop_type:peasants
								culture = scope:target
							}
						}
					}
				}
				AND = {
					root = pop_type:tribesmen
					scope:target_origin = {
						any_location_in_province = {
							any_pop = {
								pop_type = pop_type:tribesmen
								culture = scope:target
							}
						}
					}
				}
				AND = {
					root = pop_type:laborers
					scope:target_origin = {
						any_location_in_province = {
							any_pop = {
								pop_type = pop_type:laborers
								culture = scope:target
							}
						}
					}
				}
			}
		}
		map_color = define:NMapColors|MAP_COLOR_HIGH
	}
	
	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_destination
		name = "relocate_people_select_destination_province"
		none_available_msg_key = "relocate_people_no_provinces"
		column = {
			data = name
		}
		column = {
			data = population
		}
		column = {
			data = province_culture_percentage
		}
		default_sort = province_culture_percentage
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = introduce_cultural_settlements interaction_target = target_destination } }
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = relocate_people interaction_target = target_destination } }
			NOT = { this = scope:target_origin }
		}
		map_color = define:NMapColors|MAP_COLOR_LOW
	}

	on_activate = {
		custom_tooltip = introduce_cultural_settlements_tt
		if = {
			limit = {
				exists = scope:target_origin
				exists = scope:target_destination
			}
			add_migration = {
				owner = scope:actor
				from = scope:target_origin.province_definition
				to = scope:target_destination.province_definition
				culture = scope:target
				type = scope:target_pop
				amount = {
					value = "scope:target_origin.culture_population(scope:target)"
					multiply = 0.01
					min = 0.001
					max = 0.25
				}
				months = -1
			}
			scope:target_destination = {
				set_variable = {
					name = promoting_culture
					value = scope:target
				}
				set_variable = {
					name = no_reapplying_promotion
				}
			}
		}
	}

	on_deactivate = {
		if = {
			limit = {
				exists = scope:target_origin
				exists = scope:target_destination
			}
			remove_migration = {
				owner = scope:actor
				from = scope:target_origin.province_definition
				to = scope:target_destination.province_definition
				culture = scope:target
				type = scope:target_pop
			}
			scope:target_destination = {
				remove_variable = promoting_culture
				remove_variable = no_reapplying_promotion
			}
			scope:target_destination = {
				set_variable = {
					name = promoting_culture
					value = scope:target_destination.dominant_culture
				}
			}
		}
	}
	
	is_finished = {
		trigger_if = {
			limit = {
				exists = scope:target_origin
				exists = scope:target_destination
			}
				scope:target_origin = {
					culture_population = {
						culture = scope:target
						value < 1
					}
				}
		}
	}

	progress = {
		scope:target_origin = {
			value = "culture_population(scope:target)"
		}
		scope:target_destination = {
			value = "culture_population(scope:target)"
		}
	}

	map_marker = {
		show_for_owner = yes
	}

}
