relocate_people = {
	ability = adm

	# Several copies can run, each from its own origin to its own destination, each with its own cost
	# (icd_cabinet_multi_effects.txt).
	allow_multiple = yes

	potential = {
		num_locations > 0 
		OR = {
			culture = { has_culture_group = culture_group:turkic_group }
			culture = { has_culture_group = culture_group:iranian_group }
			country_rank_level = 4
		}
	}
	
	allow = {
		cultural_unity < 1.0
		country_rank = country_rank:rank_empire
		has_embraced_institution = institution:legalism
	}
	
	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_origin
		name = "relocate_people_select_origin_province"
		none_available_msg_key = "relocate_people_no_provinces"
		column = {
			data = name
		}
		column = {
			data = population
		}
		default_sort = population
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = relocate_people interaction_target = target_origin } }
		}
		map_color = define:NMapColors|MAP_COLOR_HIGH
	}
	
	select_trigger = {
		looking_for_a = culture
		target_flag = target
		#for the AI we have an optimisation to only look through religions that are present. Human can look at all...but will have some disabled
		source_ai_override = actor
		source_flags_ai_override = include_any_present
		name = "relocate_people_select_culture"
		none_available_msg_key = "relocate_people_no_culture"
		column = {
			data = name
		}
		visible = {
			scope:target_origin = { 
				culture_population = {
					culture = root
					value > 0
				}
			}
		}
	}
	
	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_destination
		name = "relocate_people_select_destination_province"
		none_available_msg_key = "relocate_people_no_provinces"
		column = {
			data = name
		}
		column = {
			data = population
		}
		column = {
			data = province_culture_percentage
		}
		default_sort = province_culture_percentage
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = relocate_people interaction_target = target_destination } }
			NOT = { this = scope:target_origin }
		}
		map_color = define:NMapColors|MAP_COLOR_LOW
	}

	on_activate = {
		custom_tooltip = relocate_people_tt
		if = {
			limit = {
				exists = scope:target_origin
				exists = scope:target_destination
			}
			add_migration = {
				owner = scope:actor
				from = scope:target_origin.province_definition
				to = scope:target_destination.province_definition
				culture = scope:target
				amount = {
					value = scope:target_origin.population
					multiply = 0.001
					min = 0.001
					max = 0.25
				}
				months = -1
			}
			scope:target_origin = {
				add_province_modifier = {
					modifier = icd_population_removal_relocate
				}
			}
			
			scope:target_origin = {
				set_variable = {
					name = icd_relocate_cost
					value = {
						value = 1
						multiply = {
							value = "scope:target_origin.province_capital.distance_to(scope:target_destination.province_capital)"
						}
					}
				}
			}
			scope:actor = {
				if = {
					limit = { NOT = { has_variable = icd_relocate_cost_total } }
					set_variable = { name = icd_relocate_cost_total value = 0 }
				}
				change_variable = { name = icd_relocate_cost_total add = scope:target_origin.var:icd_relocate_cost }
				icd_refresh_relocation_cost = yes
			}
		}
	}

	on_deactivate = {
		if = {
			limit = {
				exists = scope:target_origin
				exists = scope:target_destination
			}
			remove_migration = {
				owner = scope:actor
				from = scope:target_origin.province_definition
				to = scope:target_destination.province_definition
				culture = scope:target
			}
			scope:target_origin = {
				remove_province_modifier = icd_population_removal_relocate
			}
			scope:actor = {
				if = {
					limit = {
						has_variable = icd_relocate_cost_total
						scope:target_origin = { has_variable = icd_relocate_cost }
					}
					change_variable = { name = icd_relocate_cost_total subtract = scope:target_origin.var:icd_relocate_cost }
				}
				icd_refresh_relocation_cost = yes
			}
			scope:target_origin = {
				remove_variable = icd_relocate_cost
			}
		}
	}
	
	is_finished = {
		trigger_if = {
			limit = {
				exists = scope:target_origin
				exists = scope:target_destination
			}
				scope:target_origin = {
					culture_population = {
						culture = scope:target
						value < 1
					}
				}
		}
	}

	progress = {
		scope:target_origin = {
			value = "culture_population(scope:target)"
		}
		scope:target_destination = {
			value = "culture_population(scope:target)"
		}
	}

	map_marker = {
		show_for_owner = yes
	}

	ai_will_do = {
		value = -1000
	}
}
