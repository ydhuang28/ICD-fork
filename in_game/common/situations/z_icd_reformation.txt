		name = "reformation_confessional_settlement"
		trigger = {
			AND = {
				situation:council_of_trent = {
					situation_is_active = no
					situation_has_ended = yes
				}
				situation:war_of_religions = {
					situation_is_active = no
					situation_has_ended = yes
				}
			}
		}
	}
	outcome = {
		name = "reformation_confessional_age_passes"
		trigger = {
			current_year >= 1700
		}
	}
	
	visible = {
		exists = religion
		OR = {
			AND = {
				religion.group = religion_group:christian

						id = icd_radical_reformation.1
						years = 5
					}

				}
