REPLACE:enforce_culture = {

	type = subject
	category = CATEGORY_SUBJECT_ACTIONS

	potential = {
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_subject = {
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info_subject
		}
		visible = {
		}
        enabled = {
			subject_loyalty >= MIN_SUBJECT_LOYALTY_FOR_ACTION
			liberty_desire < MAX_SUBJECT_LIBERTY_DESIRE_FOR_ACTION
		}
	}
	
	
	select_trigger = {
		looking_for_a = culture
		interaction_source_list = {
			scope:actor = {
				every_primary_or_accepted_or_tolerated_culture = {
					add_to_list = source
				}
			}
			scope:recipient = {
				every_present_culture_in_country = {
					add_to_list = source
				}
			}
		}
		target_flag = target_culture
		name = "choose_culture"
		column = {
			data = name
		}
		visible = {
			scope:recipient = {
				OR = {
					AND = {
						culture_percentage_in_country = {
							culture = root
							value > 0.1
						}
						scope:actor = {
							is_ai = no
						}
					}
					AND = {
						is_subject_type = colonial_nation
						root = scope:actor.culture
					}
					AND = {
						scope:actor = {
							is_ai = yes
						}
						root = { is_primary_in = scope:actor }
					}
				}
				NOT = {
					root = scope:recipient.culture
				}
			}
		}
	}

	effect = {
		scope:recipient = {
				change_culture = scope:target_culture
				every_cabinet = {
				limit = {
					has_cabinet_action = yes
					or = {
						cabinet_action = cabinet_action:assimilate_area
						cabinet_action = cabinet_action:promote_culture
						cabinet_action = cabinet_action:promote_culture_edited
					}
				}
				stop_cabinet_action = {}
			}
		}
		scope:recipient = {
			add_liberty_desire = liberty_desire_ultimate_plus
		}
	}
}
