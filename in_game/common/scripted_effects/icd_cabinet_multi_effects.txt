# Cabinet actions that can run more than once at a time.

# Country scope. icd_relocation_effort_cost sized by the distances of every running Relocate People: each origin
# province keeps its own distance in icd_relocate_cost, and the country sums them in icd_relocate_cost_total.
icd_refresh_relocation_cost = {
	if = {
		limit = {
			has_variable = icd_relocate_cost_total
			var:icd_relocate_cost_total > 0
		}
		add_country_modifier = {
			modifier = icd_relocation_effort_cost
			size = var:icd_relocate_cost_total
			mode = replace
		}
	}
	else = {
		remove_country_modifier = icd_relocation_effort_cost
		remove_variable = icd_relocate_cost_total
	}
}
