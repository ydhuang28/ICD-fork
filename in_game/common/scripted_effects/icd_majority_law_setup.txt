
icd_majority_law_setup = {
	
	every_country = {
		limit = { 
			law:cultural_majority_law = { is_implementable_in = prev } 
			NOT = {
				has_policy = minority_rule
				has_policy = dialects_of_german
				has_policy = dialects_of_arabic
				has_policy = ethnic_communities
				has_policy = regional_autonomy
				has_policy = monocultural_opportunity
				has_policy = dialects_of_italian
				has_policy = dialects_of_tatar
				has_policy = dialects_of_mandarin
				has_policy = gradual_assimilation
			}
		}
		if = {
			limit = {
				policy:minority_rule = {
					is_implementable_in = prev 
				}
			}
			add_policy = policy:minority_rule
		}
		else_if = {
			limit = {
				policy:dialects_of_tatar = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:dialects_of_tatar
		}
		else_if = {
			limit = {
				policy:dialects_of_mandarin = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:dialects_of_mandarin
		}
		else_if = {
			limit = {
				policy:dialects_of_german = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:dialects_of_german
		}
		else_if = {
			limit = {
				policy:dialects_of_arabic = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:dialects_of_arabic
		}
		else_if = {
			limit = {
				policy:dialects_of_english = {
					is_implementable_in = prev
				}
				NOT = {
					tag = SCO
					culture = culture:scottish
				}
			}
			add_policy = policy:dialects_of_english
		}
		else_if = {
			limit = {
				policy:gradual_assimilation = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:gradual_assimilation
		}
	}
	

}
