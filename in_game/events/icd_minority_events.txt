namespace = icd_minority_events

icd_minority_events.1000 = { ###expel minority cabinet action###
	hide_portraits = yes
	type = country_event
	title = icd_minority_events.1.title
	desc = icd_minority_events.1.desc
	outcome = neutral

	hidden = yes
	
	illustration_tags = {
        10 = angry
        10 = exterior
    }
	immediate = {
		event_illustration_poptype_effect = { foreground = pop_type:soldiers background = pop_type:soldiers }
		
		var:icd_expel_culture_people = {
			save_scope_as = icd_expel_scope
		}
		
		ordered_country_in_diplomatic_range = { ##check if has primary country of necessary size###
			limit = {
				culture = scope:icd_expel_scope
				NOT = { 
					this = root
					any_overlord_or_above = { 
						this = root
					}
				}
				num_provinces >= 3
				total_population > icd_country_doubled_population_culture
			}
			order_by = "border_distance_to(root)"
			check_range_bounds = no
			max = 1
			save_scope_as = target_recipient_country
		}

		if = { ##check if has accepted country###
			limit = {
				NOT = {
					exists = scope:target_recipient_country
				}
			}
			ordered_country_in_diplomatic_range = {
				limit = {
					num_provinces >= 3
					has_primary_or_accepted_culture = scope:icd_expel_scope
					NOT = { 
						this = root
						any_overlord_or_above = { 
							this = root
						}
					}
					total_population > icd_country_doubled_population_culture
				}
				order_by = "border_distance_to(root)"
				check_range_bounds = no
				max = 1
				save_scope_as = target_recipient_country
			}
		}
		
		if = { ##check if has kindred opinion or shared culture group###
			limit = {
				NOT = {
					exists = scope:target_recipient_country
				}
			}
			ordered_country_in_diplomatic_range = {
				limit = {
					num_provinces >= 3
					culture = {
						cultural_view = {
							 target = scope:icd_expel_scope
							 value = kindred
						}
						has_shared_culture_group = scope:icd_expel_scope
					}
					total_population > icd_country_doubled_population_culture
					NOT = { 
						this = root
						any_overlord_or_above = { 
							this = root
						}
					}
				}
				order_by = "border_distance_to(root)"
				check_range_bounds = no
				max = 1
				save_scope_as = target_recipient_country
			}
		}
		
		if = { ##check for smaller prinary country ###
			limit = {
				NOT = {
					exists = scope:target_recipient_country
				}
			}
			ordered_country_in_diplomatic_range = {
				limit = {
					culture = scope:icd_expel_scope
					NOT = { 
						this = root
						any_overlord_or_above = { 
							this = root
						}
					}
					num_provinces >= 3
				}
				order_by = total_population
				check_range_bounds = no
				max = 1
				save_scope_as = target_recipient_country
			}
		}
		
		if = {##jewish check###
			limit = {
				NOT = {
					exists = scope:target_recipient_country
				}
				scope:icd_expel_scope = {
					has_culture_group = culture_group:jewish_group
				}
			}
			ordered_country_in_diplomatic_range = {
				limit = {
					total_population > icd_country_doubled_population
					num_provinces >= 3
					NOT = { 
						this = root
						any_overlord_or_above = { 
							this = root
						}
					}
					OR = {
						has_policy = icd_jewish_charter
						has_policy = council_of_three_lands_policy
						has_policy = icd_subjugated_communities
						modifier:tolerance_heathen > 0.5
						AND = {
							country_has_estate = estate_type:dhimmi_estate
							any_pop = {
								is_dhimmi = yes
								religion = religion:judaism
							}
						}
					}
				}
				order_by = "border_distance_to(root)"
				check_range_bounds = no
				max = 1
				save_scope_as = target_recipient_country
			}
		}
		
	}

	option = { #####
		name = icd_minority_events.1.a
		
		if = {#####
			limit = {
				exists = scope:target_recipient_country
				scope:target_recipient_country = {
					NOT = {
						has_variable = icd_expel_origin_country
					}
				}
			}
			scope:target_recipient_country = {
				trigger_event_silently = icd_minority_events.1001
				set_variable = {
					name = icd_expel_origin_country
					value = root
				}	
			}
		}
		else = {#####
			trigger_event_silently = icd_minority_events.1002
		}
		
	}
}

icd_minority_events.1003 = { ###begin expulsion###
	hide_portraits = yes
	type = country_event
	title = icd_minority_events.1003.title
	desc = icd_minority_events.1003.desc
	outcome = neutral

	
	illustration_tags = {
        10 = angry
        10 = exterior
    }
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
	}

	option = { #####
		name = icd_minority_events.1003.a
		
		var:icd_expel_destination_country = {
			save_scope_as = target_recipient_country
		}
		
		scope:target_recipient_country = {
			capital.province = {
				save_scope_as = target_destination
			}
			random_province = {
				limit = { NOT = { this = scope:target_destination }
					NOT = { province_definition = { any_location_in_province_definition = { owner ?= root } }}
				}
				save_scope_as = target_destination_two
			}
			random_province = {
				limit = { NOT = { this = scope:target_destination }
					NOT = { this = scope:target_destination_two }
					NOT = { province_definition = { any_location_in_province_definition = { owner ?= root } }}
				}
				save_scope_as = target_destination_three
			}
		}
		
		var:icd_expel_culture_people = {
			save_scope_as = icd_migration_culture_scope
		}
		
		var:icd_expel_culture_people = {
			change_cultural_view = {
				target = root.culture
				change = -3
			}
		}
		
		add_stability = {
			value = {
				value = -1
				multiply = "culture_percentage_in_country(scope:icd_migration_culture_scope)"
				multiply = 200
			}
		}
		add_gold = {
			value = {
				value = 1
				multiply = {
					value = "culture_population_in_country(scope:icd_migration_culture_scope)"
					max = 1000
				}
			}
		}
		
		custom_tooltip = icd_begin_expelling_tt
		hidden_effect = {
			every_province = {
				add_migration = {
					from = this.province_definition
					to = scope:target_destination.province_definition
					culture = scope:icd_migration_culture_scope
					amount = {
						value = 0.500
					}
					months = 24
				}
			}
			every_province = {
				add_migration = {
					from = this.province_definition
					to = scope:target_destination_two.province_definition
					culture = scope:icd_migration_culture_scope
					amount = {
					value = 0.500
					}
					months = 24
				}
			}
			every_province = {
				add_migration = {
					from = this.province_definition
					to = scope:target_destination_three.province_definition
					culture = scope:icd_migration_culture_scope
					amount = {
					value = 0.500
					}
					months = 24
				}
			}
		}
		
		remove_variable = icd_expel_culture_people
		remove_variable = icd_expel_destination_country
	}
	
	option = { ##on second thought###
		name = icd_minority_events.1003.b
		
		remove_variable = icd_expel_culture_people
		remove_variable = icd_expel_destination_country
	}
}

icd_minority_events.1 = { ###jewish exodus###
	hide_portraits = yes
	type = country_event
	title = icd_minority_events.1.title
	desc = icd_minority_events.1.desc
	outcome = neutral

	trigger = {
		"religion_percentage_in_country(religion:judaism)" > 0
		any_owned_location = {
			any_pop = {
				pop_size > 0
				religion = religion:judaism
				is_protected_pop = no
			}
		}
		NOT = {
			has_policy = icd_jewish_charter
			has_policy = council_of_three_lands_policy
			has_policy = icd_aljama_communities
			has_policy = icd_jewish_ghetto
			modifier:tolerance_heathen > 0
		}
		capital.continent = continent:europe
		OR = {
			has_variable = icd_israelite_dhimmi_disabled
			NOT = { religion.group = religion_group:muslim }
		}
		NOT = {
			has_variable = icd_jewish_banned
		}
	}
	illustration_tags = {
        10 = angry
        10 = interior
    }
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
		
		icd_jewish_migration_save_region = yes
	}
	
	option = { ##reinstate their privileges###
		name = icd_minority_events.1.a
		
		trigger = {
			OR = {
				religion.group = religion_group:christian
				religion.group = religion_group:muslim
			}
		}
		
		if = {
			limit = { religion.group = religion_group:christian }
			add_policy = policy:icd_jewish_charter
		}
		if = {
			limit = { religion.group = religion_group:muslim }
			remove_variable = icd_israelite_dhimmi_disabled
		}
		
		set_variable = {
			name = icd_jewish_permitted
			years = 5
		}
	}
	option = {
		name = icd_minority_events.1.b
		
		set_variable = {
			name = icd_jewish_banned
			years = 55
		}
		set_variable = {
			name = icd_jewish_migration
			months = 12
		}
		
		icd_jewish_migration_perform = yes
	}
	option = {
		name = icd_minority_events.1.c
		trigger = {
			OR = {
				current_year > 1570
				has_global_variable = icd_cum_nimis_absurdum
			}
			religion.group = religion_group:christian
		}
		
		remove_policy = policy:icd_jewish_charter
		add_policy = policy:icd_jewish_ghetto
		set_variable = {
			name = icd_jewish_permitted
			years = 5
		}
	}
}
