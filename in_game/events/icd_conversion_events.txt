namespace = icd_conversion_events

icd_conversion_events.1000 = { # Voluntary Conversions##
	type = country_event
	hide_portraits = yes
	title = icd_conversion_events.1000.title
	desc = icd_conversion_events.1000.desc
	outcome = neutral
		
	trigger = {
		NOT = { has_game_rule = icd_religious_conversion_vanilla }
		
		NOT = {
			has_policy = icd_ethnic_religion
			has_policy = icd_custodians
		}
		OR = {
			icd_is_pagan = no
			is_ai = no
		}
	}
	
	hidden = yes

	illustration_tags = {
	  	10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		
		
		every_owned_location = {
			limit = {
				NOT = {
					has_province_modifier = icd_promote_religion_province_modifier
				}
				OR = {
					integration_level = core
					integration_level = integrated
				}
				
			}
			every_pop = {
				limit = { 
					religion != root.religion
					NOT = { pop_type = pop_type:slaves }
					NOT = { pop_type = pop_type:clergy }
					NOT = { pop_type = pop_type:tribesmen }
					location = {
						local_control >= 0.1
					}
					is_protected_pop = yes
				}
				split_pop = {
					fraction = {
						value = icd_voluntary_conversion_rate
					}
					religion = root.religion
				}
			}
		}

		
		every_pop = { ##remove the small minorities###
			limit = {
				location = {
					religion_population = {
						religion = prev.religion
						value < 0.05
					}
				}
				religion != root.religion
				religion.group != religion_group:israelite_group
				religion != religion:jain
			}
			change_pop_religion = location.dominant_religion
		}
	}

	option = {
		name = icd_conversion_events.1000.desc
	}
}

icd_conversion_events.999 = { # Coercive Conversions##
	type = country_event
	hide_portraits = yes
	title = icd_conversion_events.1000.title
	desc = icd_conversion_events.1000.desc
	outcome = neutral
		
	trigger = {
		any_province = {
			has_province_modifier = icd_promote_religion_province_modifier
		}
		NOT = { has_game_rule = icd_religious_conversion_vanilla }
	}


	illustration_tags = {
	  	10 = happy
		10 = interior
	}
	
	hidden = yes

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }

		# Fork: a promotion whose religion is gone leaves only the penalty. The AI's 48-month variable can lapse before a
		# modifier that add_and_extend has extended, and the conversion below read the missing variable for every pop.
		every_province = {
			limit = {
				has_province_modifier = icd_promote_religion_province_modifier
				NOT = { has_variable = promoting_religion }
			}
			remove_province_modifier = icd_promote_religion_province_modifier
		}

		every_province = {
			limit = {
				has_province_modifier = icd_promote_religion_province_modifier
			}
			every_location_in_province = {
				every_pop = {
					limit = {
						religion != location.province.var:promoting_religion
						owner ?= root
						is_protected_pop = no
						pop_type != pop_type:slaves
					}
					if = {
						limit = {
							NOT = { owner ?= { can_see_situation = situation:reformation } }
							religion.group = religion_group:christian
						}
						add_pop_satisfaction = {
							value = pop_satisfaction
							multiply = -1
						}
					}
					split_pop = {
						size = icd_coercive_conversion_rate
						religion = location.province.var:promoting_religion
					}
				}
			}
		}
	}

	option = {
		name = icd_conversion_events.1000.desc
	
		
	}
}

icd_conversion_events.900 = { # Communal COnversions##
	type = country_event
	hide_portraits = yes
	title = icd_conversion_events.1000.title
	desc = icd_conversion_events.1000.desc
	outcome = neutral
		
	trigger = {
		NOT = { has_game_rule = icd_religious_conversion_vanilla }
	}

	hidden = yes

	illustration_tags = {
	  	10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }	
		if = {
			limit = {
				has_policy = force_slave_conversion
			}
			every_pop = {
				limit = {
					pop_type = pop_type:slaves
				}
				split_pop = {
					religion = root.religion
					size = {
						value = 0.3
					}
				}
			}
		}
		###Convert Pagan###
		if = {
			limit = {
				NOT = {
					has_policy = icd_ethnic_religion
					has_policy = icd_custodians
					has_policy = icd_dual_faiths
				}
				OR = {
					icd_is_pagan = no
					is_ai = no
				}
			}
			every_pop = {
				limit = {
					icd_is_pagan = yes
					is_protected_pop = no
					pop_type != pop_type:slaves
					owner ?= root
					location = {
						"religion_percentage(root.religion)" >= 0.2
						NOT = {
							has_location_modifier = icd_religious_zeal
						}
					}
				}
				split_pop = {
					religion = root.religion
					size = {
						value = icd_communal_conversion_rate_pagan
					}
				}
			}
		}
		
		if = { ###Culture Conversion###
			limit = {
				NOT = {
					has_policy = icd_secular_state
					has_policy = icd_yassa_tolerance
					has_policy = icd_sulh_i_kul
				}
				OR = {
					AND = {
						has_variable = icd_confessional_culture
						root.culture = var:icd_confessional_culture
					}
					icd_is_pagan = yes
					has_policy = icd_ethnic_religion
					has_policy = icd_custodians
				}
				NOT = {
					is_subject_type = fiefdom
					is_subject_type = dominion
					is_subject_type = appanage
					AND = {
						has_variable = icd_enforced_religion
						religion = var:icd_enforced_religion
					}
				}
			}
			if = {
				limit = {
					NOT = {
						root.culture = {
							has_culture_group = culture_group:jewish_group
						}
					}
				}
				every_pop = {
					limit = { 
						culture = root.culture
						religion != root.religion
						owner ?= root
						pop_type != pop_type:slaves
						location = {
							NOT = {
								has_location_modifier = icd_religious_zeal
							}
						}
					}
					split_pop = {
						religion = root.religion
						size = {
							value = icd_communal_conversion_rate
						}
					}
				}
			}
			else_if = {
				limit = {
					root.culture = {
						has_culture_group = culture_group:jewish_group
					}
				}
				every_pop = {
					limit = {
						religion != root.religion
						owner ?= root
						pop_type != pop_type:slaves
						location = {
							NOT = {
								has_location_modifier = icd_religious_zeal
							}
						}
						culture = { has_shared_culture_group = root.culture }
					}
					split_pop = {
						religion = root.religion
						size = {
							value = icd_communal_conversion_rate
						}
					}
				}
			}
		}
		
		
		if = {
			limit = {
				has_variable = icd_bosnian_conversion
			}
			every_pop = {
				limit = {
					culture = culture:bosnian
					religion != root.religion
				}
				split_pop = {
					religion = root.religion
					size = {
						value = 0.2
					}
				}
			}
		}
	}

	option = {
		
	}
}
